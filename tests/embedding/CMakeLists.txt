cmake_minimum_required(VERSION 3.25)
project(MCPHostTargetConsumer LANGUAGES CXX)
set(UNIMEMORY_BACKEND standard CACHE STRING "Backend")
set(UNIMEMORY_STATISTICS OFF CACHE BOOL "Statistics")
set(UNIMEMORY_CHECKS AUTO CACHE STRING "Checks")
set(UNIMEMORY_BUILD_TESTS OFF CACHE BOOL "No allocator tests")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../external/unimemory" official-unimemory)
option(MCP_EMBED_SOURCE "Exercise add_subdirectory instead of an installed package" OFF)
if(MCP_EMBED_SOURCE)
    # The library must preserve the host's configuration set and build only its target.
    set(CMAKE_CONFIGURATION_TYPES "Debug;Release;HostCustom" CACHE STRING "Host configurations" FORCE)
    foreach(kind IN ITEMS CXX_FLAGS EXE_LINKER_FLAGS SHARED_LINKER_FLAGS MODULE_LINKER_FLAGS STATIC_LINKER_FLAGS)
        set(CMAKE_${kind}_HOSTCUSTOM "${CMAKE_${kind}_RELEASE}" CACHE STRING "Host custom configuration")
    endforeach()
    set(CMAKE_MAP_IMPORTED_CONFIG_HOSTCUSTOM Release)
    add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../.." mcp)
    if(MCP_BUILD_TESTS OR MCP_BUILD_EXAMPLES OR MCP_BUILD_CLI)
        message(FATAL_ERROR "Embedding unexpectedly enabled MCP development targets")
    endif()
    if(NOT CMAKE_CONFIGURATION_TYPES STREQUAL "Debug;Release;HostCustom")
        message(FATAL_ERROR "MCP changed its host's build configurations")
    endif()
    foreach(unwanted IN ITEMS test_mcp exe_cli example_features)
        if(TARGET ${unwanted})
            message(FATAL_ERROR "Embedding unexpectedly created ${unwanted}")
        endif()
    endforeach()
else()
    find_package(MCP 1.0.0 CONFIG REQUIRED)
endif()
add_executable(mcp_host_consumer ../consumer/main.cpp)
target_link_libraries(mcp_host_consumer PRIVATE MCP::MCP)
