cmake_minimum_required(VERSION 3.25)
project(MCPHostTargetConsumer LANGUAGES CXX)
set(UNIMEMORY_BACKEND standard CACHE STRING "Backend")
set(UNIMEMORY_STATISTICS OFF CACHE BOOL "Statistics")
set(UNIMEMORY_CHECKS AUTO CACHE STRING "Checks")
set(UNIMEMORY_BUILD_TESTS OFF CACHE BOOL "No allocator tests")
add_subdirectory("${CMAKE_CURRENT_LIST_DIR}/../../external/unimemory" official-unimemory)
find_package(MCP 0.0.1 CONFIG REQUIRED)
add_executable(mcp_host_consumer ../consumer/main.cpp)
target_link_libraries(mcp_host_consumer PRIVATE MCP::MCP)
