cmake_minimum_required(VERSION 3.25)
project(MCPInstalledConsumer LANGUAGES CXX)
find_package(MCP 1.0.0 CONFIG REQUIRED)
add_executable(mcp_consumer main.cpp)
target_link_libraries(mcp_consumer PRIVATE MCP::MCP)

set(_mcp_readme_source "${CMAKE_CURRENT_SOURCE_DIR}/readme.cpp")
if(EXISTS "${CMAKE_CURRENT_BINARY_DIR}/compiled_readme.cpp")
    set(_mcp_readme_source "${CMAKE_CURRENT_BINARY_DIR}/compiled_readme.cpp")
endif()
add_executable(mcp_readme "${_mcp_readme_source}")
target_link_libraries(mcp_readme PRIVATE MCP::MCP)
if(EXISTS "${CMAKE_CURRENT_BINARY_DIR}/compiled_chinese_readme.cpp")
    add_library(mcp_chinese_readme OBJECT "${CMAKE_CURRENT_BINARY_DIR}/compiled_chinese_readme.cpp")
    target_link_libraries(mcp_chinese_readme PRIVATE MCP::MCP)
endif()
if(EXISTS "${CMAKE_CURRENT_BINARY_DIR}/documented_examples.cpp")
    add_library(mcp_documented_examples OBJECT "${CMAKE_CURRENT_BINARY_DIR}/documented_examples.cpp")
    target_link_libraries(mcp_documented_examples PRIVATE MCP::MCP)
endif()
