cmake_minimum_required(VERSION 3.25)
set(VCPKG_APPLOCAL_DEPS OFF)
option(MCP_BUILD_TESTS "Build the retained MCP test suite" ON)
option(MCP_BUILD_EXAMPLES "Build stdio and HTTP examples" ON)
option(MCP_BUILD_CLI "Build the MCP command-line client/server" ON)
option(MCP_BUILD_FUZZERS "Build Clang libFuzzer entry points" OFF)
option(MCP_COVERAGE "Instrument GCC/Clang builds for coverage" OFF)
set(MCP_SANITIZER "" CACHE STRING "address or thread (GCC/Clang)")
set(MCP_UNIMEMORY_SOURCE_DIR "${CMAKE_CURRENT_LIST_DIR}/external/unimemory" CACHE PATH "Official UniMemory source fallback")
include(cmake/Dependencies.cmake)
include(cmake/Memory.cmake)
mcp_configure_memory_profile()
if(MCP_BUILD_TESTS)
    list(APPEND VCPKG_MANIFEST_FEATURES tests)
endif()
if(UNIMEMORY_BACKEND STREQUAL "mimalloc")
    list(APPEND VCPKG_MANIFEST_FEATURES mimalloc)
    list(APPEND VCPKG_OVERLAY_TRIPLETS "${MCP_UNIMEMORY_SOURCE_DIR}/cmake/triplets")
elseif(UNIMEMORY_BACKEND STREQUAL "jemalloc" AND (CMAKE_HOST_WIN32 OR CMAKE_SYSTEM_NAME STREQUAL "Windows"))
    list(APPEND VCPKG_MANIFEST_FEATURES jemalloc)
endif()
project(MCP VERSION 0.0.1 LANGUAGES CXX)
if(CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_CONFIGURATION_TYPES "Debug;Release" CACHE STRING "Build configurations" FORCE)
endif()
set_property(GLOBAL PROPERTY USE_FOLDERS ON)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(cmake/StageRuntime.cmake)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(MCP_SANITIZER)
    if(MSVC OR NOT MCP_SANITIZER MATCHES "^(address|thread)$")
        message(FATAL_ERROR "MCP_SANITIZER requires GCC/Clang and address or thread")
    endif()
    if(MCP_SANITIZER STREQUAL "address")
        add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
        add_link_options(-fsanitize=address,undefined -fno-sanitize-recover=undefined)
    else()
        add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
        add_link_options(-fsanitize=thread)
    endif()
endif()
if(MCP_COVERAGE)
    if(MSVC OR NOT CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
        message(FATAL_ERROR "MCP_COVERAGE requires GCC/Clang")
    endif()
    add_compile_options(--coverage -O0 -g)
    add_link_options(--coverage)
endif()
set(_mcp_parent_folder "${CMAKE_FOLDER}")
set(CMAKE_FOLDER "Third party/UniMemory")
mcp_add_memory()
set(CMAKE_FOLDER "${_mcp_parent_folder}")
unset(_mcp_parent_folder)
mcp_configure_memory_profile()
include(cmake/MCP.cmake)
if(MCP_BUILD_CLI)
    include(cmake/CLI.cmake)
endif()
if(MCP_BUILD_EXAMPLES OR (MCP_BUILD_TESTS AND BUILD_TESTING))
    add_subdirectory(examples/mcp)
endif()
if(MCP_BUILD_TESTS AND BUILD_TESTING)
    add_subdirectory(tests/mcp)
endif()
include(cmake/Runtime.cmake)
include(cmake/DependencyLicenses.cmake)
mcp_install_dependency_licenses()
get_property(MCP_JSON_VERSION GLOBAL PROPERTY MCP_JSON_VERSION)
configure_package_config_file(cmake/MCPConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/MCPConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MCP)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/MCPConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMinorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/MCPConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/MCPConfigVersion.cmake"
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/MCP)
install(FILES LICENSE NOTICE PROVENANCE.md DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/MCP)
install(FILES licenses/mcp.txt DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/MCP/mcp RENAME LICENSE)
install(FILES tests/mcp/fixtures/licenses/MCP-LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/MCP/schema)
install(FILES README.md README.zh-CN.md DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/MCP)
install(FILES LICENSE NOTICE PROVENANCE.md CHANGELOG.md CONTRIBUTING.md SECURITY.md
    DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/MCP)
install(DIRECTORY docs DESTINATION ${CMAKE_INSTALL_DATADIR}/doc/MCP)
mcp_stage_runtime()

if(TARGET exe_cli)
    set_property(DIRECTORY PROPERTY VS_STARTUP_PROJECT exe_cli)
endif()
